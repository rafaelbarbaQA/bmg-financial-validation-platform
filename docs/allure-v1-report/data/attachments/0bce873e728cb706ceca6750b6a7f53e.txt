SET NOCOUNT ON; DECLARE @EventosInseridos TABLE (Id_EventoPagamento int NOT NULL); INSERT INTO EventosExternosPagamentos (Banco, DataMovimento, DataPagamento, NossoNumero, ValorPago, NumeroDocumento, Cartao, Id_MovimentoPagamentoPendente, DefasagemCobranca, Id_Autorizacao, Id_Conta, Id_Operacao, DataEntradaPagamento, DataVencimentoPadrao, DataVencimentoReal, RestituicaoSaque, Status) OUTPUT INSERTED.Id_EventoPagamento INTO @EventosInseridos (Id_EventoPagamento) SELECT PROD.Banco, ?, ?, ?, ?, NULL, CARTAO.Cartao, NULL, NULL, ?, EC.Id_Conta, ?, ?, CONVERT(char(10), EC.ProximoVencimentoPadrao, 103), EC.ProximoVencimentoReal, ?, ? FROM EstadosContas EC INNER JOIN Produtos PROD ON PROD.Id_Produto = EC.Id_Produto CROSS APPLY (SELECT TOP (1) C.Cartao FROM Cartoes C WHERE C.Id_Conta = EC.Id_Conta AND C.Portador = 1 ORDER BY C.SequencialCartao DESC) CARTAO WHERE EC.Id_Conta = ?; SELECT Id_EventoPagamento AS ID_EVENTO_PAGAMENTO FROM @EventosInseridos;
